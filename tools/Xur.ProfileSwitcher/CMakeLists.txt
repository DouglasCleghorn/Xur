cmake_minimum_required(VERSION 3.20)
project(XurProfileSwitcher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network DBus)
find_package(KF6GlobalAccel REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(INPUT REQUIRED IMPORTED_TARGET libevdev libudev)
add_executable(xur-profile-switcher main.cpp)
target_compile_options(xur-profile-switcher PRIVATE -Wall -Wextra -Werror)
target_link_libraries(xur-profile-switcher PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus KF6::GlobalAccel PkgConfig::INPUT)
set_target_properties(xur-profile-switcher PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
